   103→function formatDate(dateStr: string): string {
   104→  const d = new Date(dateStr);
   105→  const y = d.getFullYear();
   106→  const m = String(d.getMonth() + 1).padStart(2, '0');
   107→  const day = String(d.getDate()).padStart(2, '0');
   108→  return `${y}/${m}/${day}`;
   109→}
   110→
   111→function formatDateInput(dateStr: string): string {
   112→  const d = new Date(dateStr);
   113→  const y = d.getFullYear();
   114→  const m = String(d.getMonth() + 1).padStart(2, '0');
   115→  const day = String(d.getDate()).padStart(2, '0');
   116→  return `${y}-${m}-${day}`;
   117→}

   119→function formatCurrency(amount: number): string {
   120→  return amount.toLocaleString('ar-MA') + ' د.م.';

   148→function computeTaxValues(htAmount: number, tvaRate: number, taxeProfRate: number, timbreFiscal: number) {
   149→  const tvaAmount = htAmount * (tvaRate / 100);
   150→  const taxeProfAmount = htAmount * (taxeProfRate / 100);
   151→  const ttcAmount = htAmount + tvaAmount + taxeProfAmount + timbreFiscal;
   152→  return { tvaAmount, taxeProfAmount, ttcAmount };
   153→}
