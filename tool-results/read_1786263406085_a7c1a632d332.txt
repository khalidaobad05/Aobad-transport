   143→   143→// ========== Client Autocomplete Input ==========
   144→   144→function ClientAutocomplete({
   145→   145→  value,
   146→   146→  onChange,
   147→   147→  clients,
   148→   148→}: {
   149→   149→  value: string;
   150→   150→  onChange: (clientId: string, clientName: string) => void;
   151→   151→  clients: Client[];
   152→   152→}) {
   153→   153→  const [inputValue, setInputValue] = useState(value);
   154→   154→  const [isOpen, setIsOpen] = useState(false);
   155→   155→  const [highlightIndex, setHighlightIndex] = useState(-1);
   156→   156→  const wrapperRef = useRef<HTMLDivElement>(null);
   157→   157→  const inputRef = useRef<HTMLInputElement>(null);
   158→   158→
   159→   159→  // Sync external value changes (e.g. edit mode)
   160→   160→  useEffect(() => {
   161→   161→    setInputValue(value);
   162→   162→  }, [value]);
   163→   163→
   164→   164→  // Close dropdown on outside click
   165→   165→  useEffect(() => {
   166→   166→    function handleClickOutside(e: MouseEvent) {
   167→   167→      if (wrapperRef.current && !wrapperRef.current.contains(e.target as Node)) {
   168→   168→        setIsOpen(false);
   169→   169→      }
   170→   170→    }
   171→   171→    document.addEventListener('mousedown', handleClickOutside);
   172→   172→    return () => document.removeEventListener('mousedown', handleClickOutside);
   173→   173→  }, []);
   174→   174→
   175→   175→  // Filter clients based on input
   176→   176→  const filteredClients = inputValue.trim()
   177→   177→    ? clients.filter((c) =>
   178→   178→        c.name.toLowerCase().includes(inputValue.trim().toLowerCase())
   179→   179→      )
   180→   180→    : clients;
   181→   181→
   182→   182→  // Show top 8 results
   183→   183→  const shownClients = filteredClients.slice(0, 8);
   184→   184→
   185→   185→  // Check if input exactly matches an existing client
   186→   186→  const exactMatch = clients.find(
   187→   187→    (c) => c.name.toLowerCase() === inputValue.trim().toLowerCase()
   188→   188→  );
   189→   189→
   190→   190→  const isNewClient = inputValue.trim().length > 0 && !exactMatch;
   191→   191→
   192→   192→  function handleSelect(client: Client) {
   193→   193→    setInputValue(client.name);
   194→   194→    onChange(client.id, client.name);
   195→   195→    setIsOpen(false);
   196→   196→    setHighlightIndex(-1);
   197→   197→  }
   198→   198→
   199→   199→  function handleInputChange(e: React.ChangeEvent<HTMLInputElement>) {
   200→   200→    const val = e.target.value;
   201→   201→    setInputValue(val);
   202→   202→    onChange('', val); // Clear clientId while typing
   203→   203→    setIsOpen(true);
   204→   204→    setHighlightIndex(-1);
   205→   205→  }
   206→   206→
   207→   207→  function handleAddNew() {
   208→   208→    if (!inputValue.trim()) return;
   209→   209→    // clientId stays empty - will be resolved on submit
   210→   210→    onChange('', inputValue.trim());
   211→   211→    setIsOpen(false);
   212→   212→  }
   213→   213→
   214→   214→  function handleKeyDown(e: React.KeyboardEvent) {
   215→   215→    const totalItems = shownClients.length + (isNewClient ? 1 : 0);
   216→   216→
   217→   217→    if (e.key === 'ArrowDown') {
   218→   218→      e.preventDefault();
   219→   219→      setHighlightIndex((prev) => (prev < totalItems - 1 ? prev + 1 : 0));
   220→   220→    } else if (e.key === 'ArrowUp') {
   221→   221→      e.preventDefault();
   222→   222→      setHighlightIndex((prev) => (prev > 0 ? prev - 1 : totalItems - 1));
   223→   223→    } else if (e.key === 'Enter') {
   224→   224→      e.preventDefault();
   225→   225→      if (highlightIndex >= 0 && highlightIndex < shownClients.length) {
   226→   226→        handleSelect(shownClients[highlightIndex]);
   227→   227→      } else if (highlightIndex === shownClients.length && isNewClient) {
   228→   228→        handleAddNew();
   229→   229→      }
   230→   230→    } else if (e.key === 'Escape') {
   231→   231→      setIsOpen(false);
   232→   232→    } else if (e.key === 'Tab') {
   233→   233→      // If there's exactly one match, select it
   234→   234→      if (shownClients.length === 1 && !isNewClient) {
   235→   235→        e.preventDefault();
   236→   236→        handleSelect(shownClients[0]);
   237→   237→      } else if (exactMatch) {
   238→   238→        e.preventDefault();
   239→   239→        handleSelect(exactMatch);
   240→   240→      }
   241→   241→    }
   242→   242→  }
   243→   243→
   244→   244→  return (
   245→   245→    <div ref={wrapperRef} className="relative">
   246→   246→      <div className="relative">
   247→   247→        <Input
   248→   248→          ref={inputRef}
   249→   249→          type="text"
   250→   250→          value={inputValue}
   251→   251→          onChange={handleInputChange}
   252→   252→          onFocus={() => setIsOpen(true)}
   253→   253→          onKeyDown={handleKeyDown}
   254→   254→          placeholder="اكتب اسم الزبون..."
   255→   255→          className="w-full pr-8"
   256→   256→          autoComplete="off"
   257→   257→        />
   258→   258→        {inputValue && (
   259→   259→          <button
   260→   260→            type="button"
   261→   261→            onClick={() => {
   262→   262→              setInputValue('');
   263→   263→              onChange('', '');
   264→   264→              inputRef.current?.focus();
   265→   265→            }}
   266→   266→            className="absolute left-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   267→   267→          >
   268→   268→            <X className="size-3.5" />
   269→   269→          </button>
   270→   270→        )}
   271→   271→      </div>
   272→   272→
   273→   273→      {/* Dropdown */}
   274→   274→      {isOpen && (shownClients.length > 0 || isNewClient) && (
   275→   275→        <div className="absolute z-50 top-full mt-1 w-full bg-white dark:bg-gray-900 border border-gray-200 dark:border-gray-700 rounded-lg shadow-lg max-h-48 overflow-y-auto">
   276→   276→          {shownClients.map((client, idx) => (
   277→   277→            <button
   278→   278→              key={client.id}
   279→   279→              type="button"
   280→   280→              onClick={() => handleSelect(client)}
   281→   281→              onMouseEnter={() => setHighlightIndex(idx)}
   282→   282→              className={`
   283→   283→                w-full text-right px-3 py-2 text-sm flex items-center gap-2 transition-colors
   284→   284→                ${idx === highlightIndex
   285→   285→                  ? 'bg-amber-50 dark:bg-amber-900/20 text-amber-900 dark:text-amber-100'
   286→   286→                  : 'hover:bg-gray-50 dark:hover:bg-gray-800'
   287→   287→                }
   288→   288→              `}
   289→   289→            >
   290→   290→              <span className="truncate flex-1">{client.name}</span>
   291→   291→              {exactMatch?.id === client.id && (
   292→   292→                <Check className="size-3.5 text-emerald-500 shrink-0" />
   293→   293→              )}
   294→   294→            </button>
   295→   295→          ))}
   296→   296→          {isNewClient && (
   297→   297→            <button
   298→   298→              type="button"
   299→   299→              onClick={handleAddNew}
   300→   300→              onMouseEnter={() => setHighlightIndex(shownClients.length)}
   301→   301→              className={`
   302→   302→                w-full text-right px-3 py-2 text-sm flex items-center gap-2 border-t border-gray-100 dark:border-gray-800 transition-colors
   303→   303→                ${highlightIndex === shownClients.length
   304→   304→                  ? 'bg-emerald-50 dark:bg-emerald-900/20 text-emerald-700 dark:text-emerald-300'
   305→   305→                  : 'hover:bg-gray-50 dark:hover:bg-gray-800 text-emerald-600 dark:text-emerald-400'
   306→   306→                }
   307→   307→              `}
   308→   308→            >
   309→   309→              <UserPlus className="size-3.5 shrink-0" />
   310→   310→              <span>إضافة &quot;{inputValue.trim()}&quot; كزبون جديد</span>
   311→   311→            </button>
   312→   312→          )}
   313→   313→        </div>
   314→   314→      )}
   315→   315→
   316→   316→      {/* Hidden indicator: if value has clientId it's an existing client */}
   317→   317→      {!exactMatch && inputValue.trim() && (
   318→   318→        <p className="text-xs text-emerald-600 dark:text-emerald-400 mt-1 flex items-center gap-1">
   319→   319→          <UserPlus className="size-3" />
   320→   320→          سيتم إضافة &quot;{inputValue.trim()}&quot; إلى لائحة الزبائن
   321→   321→        </p>
   322→   322→      )}
   323→   323→    </div>
   324→   324→  );
   325→   325→}
