   102→function formatDate(dateStr: string): string {
   103→  const d = new Date(dateStr);
   104→  const y = d.getFullYear();
   105→  const m = String(d.getMonth() + 1).padStart(2, '0');
   106→  const day = String(d.getDate()).padStart(2, '0');
   107→  return `${y}/${m}/${day}`;
   108→}

   114→  const day = String(d.getDate()).padStart(2, '0');
   115→  return `${y}-${m}-${day}`;

   143→// ========== Client Autocomplete Input ==========
   144→function ClientAutocomplete({
   145→  value,
   146→  onChange,
   147→  clients,
   148→}: {
   149→  value: string;
   150→  onChange: (clientId: string, clientName: string) => void;
   151→  clients: Client[];
   152→}) {
   153→  const [inputValue, setInputValue] = useState(value);
   154→  const [isOpen, setIsOpen] = useState(false);
   155→  const [highlightIndex, setHighlightIndex] = useState(-1);
   156→  const wrapperRef = useRef<HTMLDivElement>(null);
   157→  const inputRef = useRef<HTMLInputElement>(null);
   158→
   159→  // Sync external value changes (e.g. edit mode)
   160→  useEffect(() => {
   161→    setInputValue(value);
   162→  }, [value]);
   163→
   164→  // Close dropdown on outside click
   165→  useEffect(() => {
   166→    function handleClickOutside(e: MouseEvent) {
   167→      if (wrapperRef.current && !wrapperRef.current.contains(e.target as Node)) {
   168→        setIsOpen(false);
   169→      }
   170→    }
   171→    document.addEventListener('mousedown', handleClickOutside);
   172→    return () => document.removeEventListener('mousedown', handleClickOutside);
   173→  }, []);
   174→
   175→  // Filter clients based on input
   176→  const filteredClients = inputValue.trim()
   177→    ? clients.filter((c) =>
   178→        c.name.toLowerCase().includes(inputValue.trim().toLowerCase())
   179→      )
   180→    : clients;
   181→
   182→  // Show top 8 results
   183→  const shownClients = filteredClients.slice(0, 8);
   184→
   185→  // Check if input exactly matches an existing client
   186→  const exactMatch = clients.find(
   187→    (c) => c.name.toLowerCase() === inputValue.trim().toLowerCase()
   188→  );
   189→
   190→  const isNewClient = inputValue.trim().length > 0 && !exactMatch;
   191→
   192→  function handleSelect(client: Client) {
   193→    setInputValue(client.name);
   194→    onChange(client.id, client.name);
   195→    setIsOpen(false);
   196→    setHighlightIndex(-1);
   197→  }
   198→
   199→  function handleInputChange(e: React.ChangeEvent<HTMLInputElement>) {
   200→    const val = e.target.value;
   201→    setInputValue(val);
   202→    onChange('', val); // Clear clientId while typing
   203→    setIsOpen(true);
   204→    setHighlightIndex(-1);
   205→  }
   206→
   207→  function handleAddNew() {
   208→    if (!inputValue.trim()) return;
   209→    // clientId stays empty - will be resolved on submit
   210→    onChange('', inputValue.trim());
   211→    setIsOpen(false);
   212→  }
   213→
   214→  function handleKeyDown(e: React.KeyboardEvent) {
   215→    const totalItems = shownClients.length + (isNewClient ? 1 : 0);
   216→
   217→    if (e.key === 'ArrowDown') {
   218→      e.preventDefault();
   219→      setHighlightIndex((prev) => (prev < totalItems - 1 ? prev + 1 : 0));
   220→    } else if (e.key === 'ArrowUp') {
   221→      e.preventDefault();
   222→      setHighlightIndex((prev) => (prev > 0 ? prev - 1 : totalItems - 1));
   223→    } else if (e.key === 'Enter') {
   224→      e.preventDefault();
   225→      if (highlightIndex >= 0 && highlightIndex < shownClients.length) {
   226→        handleSelect(shownClients[highlightIndex]);
   227→      } else if (highlightIndex === shownClients.length && isNewClient) {
   228→        handleAddNew();
   229→      }
   230→    } else if (e.key === 'Escape') {
   231→      setIsOpen(false);
   232→    } else if (e.key === 'Tab') {
   233→      // If there's exactly one match, select it
   234→      if (shownClients.length === 1 && !isNewClient) {
   235→        e.preventDefault();
   236→        handleSelect(shownClients[0]);
   237→      } else if (exactMatch) {
   238→        e.preventDefault();
   239→        handleSelect(exactMatch);
   240→      }
   241→    }
   242→  }
   243→
   244→  return (
   245→    <div ref={wrapperRef} className="relative">
   246→      <div className="relative">
   247→        <Input
   248→          ref={inputRef}
   249→          type="text"
   250→          value={inputValue}
   251→          onChange={handleInputChange}
   252→          onFocus={() => setIsOpen(true)}
   253→          onKeyDown={handleKeyDown}
   254→          placeholder="اكتب اسم الزبون..."
   255→          className="w-full pr-8"
   256→          autoComplete="off"
   257→        />
   258→        {inputValue && (
   259→          <button
   260→            type="button"
   261→            onClick={() => {
   262→              setInputValue('');
   263→              onChange('', '');
   264→              inputRef.current?.focus();
   265→            }}
   266→            className="absolute left-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   267→          >
   268→            <X className="size-3.5" />
   269→          </button>
   270→        )}
   271→      </div>
   272→
   273→      {/* Dropdown */}
   274→      {isOpen && (shownClients.length > 0 || isNewClient) && (
   275→        <div className="absolute z-50 top-full mt-1 w-full bg-white dark:bg-gray-900 border border-gray-200 dark:border-gray-700 rounded-lg shadow-lg max-h-48 overflow-y-auto">
   276→          {shownClients.map((client, idx) => (
   277→            <button
   278→              key={client.id}
   279→              type="button"
   280→              onClick={() => handleSelect(client)}
   281→              onMouseEnter={() => setHighlightIndex(idx)}
   282→              className={`
   283→                w-full text-right px-3 py-2 text-sm flex items-center gap-2 transition-colors
   284→                ${idx === highlightIndex
   285→                  ? 'bg-amber-50 dark:bg-amber-900/20 text-amber-900 dark:text-amber-100'
   286→                  : 'hover:bg-gray-50 dark:hover:bg-gray-800'
   287→                }
   288→              `}
   289→            >
   290→              <span className="truncate flex-1">{client.name}</span>
   291→              {exactMatch?.id === client.id && (
   292→                <Check className="size-3.5 text-emerald-500 shrink-0" />
   293→              )}
   294→            </button>
   295→          ))}
   296→          {isNewClient && (
   297→            <button
   298→              type="button"
   299→              onClick={handleAddNew}
   300→              onMouseEnter={() => setHighlightIndex(shownClients.length)}
   301→              className={`
   302→                w-full text-right px-3 py-2 text-sm flex items-center gap-2 border-t border-gray-100 dark:border-gray-800 transition-colors
   303→                ${highlightIndex === shownClients.length
   304→                  ? 'bg-emerald-50 dark:bg-emerald-900/20 text-emerald-700 dark:text-emerald-300'
   305→                  : 'hover:bg-gray-50 dark:hover:bg-gray-800 text-emerald-600 dark:text-emerald-400'
   306→                }
   307→              `}
   308→            >
   309→              <UserPlus className="size-3.5 shrink-0" />
   310→              <span>إضافة &quot;{inputValue.trim()}&quot; كزبون جديد</span>
   311→            </button>
   312→          )}
   313→        </div>
   314→      )}
   315→
   316→      {/* Hidden indicator: if value has clientId it's an existing client */}
   317→      {!exactMatch && inputValue.trim() && (
   318→        <p className="text-xs text-emerald-600 dark:text-emerald-400 mt-1 flex items-center gap-1">
   319→          <UserPlus className="size-3" />
   320→          سيتم إضافة &quot;{inputValue.trim()}&quot; إلى لائحة الزبائن
   321→        </p>
   322→      )}
   323→    </div>
   324→  );
   325→}

   449→  // Ensure a client exists (find or create), returns the client ID
   450→  async function ensureClient(name: string): Promise<string | null> {
   451→    try {
   452→      const res = await fetch('/api/clients/ensure', {
   453→        method: 'POST',
   454→        headers: { 'Content-Type': 'application/json' },
   455→        body: JSON.stringify({ name }),
   456→      });
   457→      if (!res.ok) return null;
   458→      const json = await res.json();
   459→      // Refresh clients list to include any newly created
   460→      if (json.created) {
   461→        fetchClients();
   462→      }
   463→      return json.data.id;
   464→    } catch {
   465→      return null;
   466→    }

   469→  async function handleSubmit(e: React.FormEvent) {
   470→    e.preventDefault();
   471→    if (!form.date) {
   472→      toast.error('التاريخ مطلوب');
   473→      return;
   474→    }
   475→    if (!form.vehicleId) {
   476→      toast.error('المركبة مطلوبة');
   477→      return;
   478→    }
   479→
   480→    // Validate orders - each must have a client name and package count
   481→    const validOrders = form.orders.filter(
   482→      (o) => o.clientName.trim() && o.packageCount > 0
   483→    );
   484→    if (validOrders.length === 0) {
   485→      toast.error('يجب إضافة طلبية واحدة صالحة على الأقل (اسم الزبون وعدد الطرود)');
   486→      return;
   487→    }
   488→
   489→    try {
   490→      setSubmitting(true);
   491→
   492→      // Step 1: Ensure all clients exist (resolve or create)
   493→      const resolvedOrders = await Promise.all(
   494→        validOrders.map(async (o) => {
   495→          let clientId = o.clientId;
   496→          // If no clientId, the client might be new - ensure it exists
   497→          if (!clientId) {
   498→            clientId = (await ensureClient(o.clientName.trim())) || '';
   499→          }
   500→          return {
   501→            clientId,
   502→            packageCount: o.packageCount,
   503→            description: o.description?.trim() || undefined,
   504→          };
   505→        })
   506→      );
   507→
   508→      // Validate all clients were resolved
   509→      const unresolved = resolvedOrders.find((o) => !o.clientId);
   510→      if (unresolved) {
   511→        throw new Error('فشل في تسجيل أحد الزبائن');
   512→      }
   513→
   514→      // Step 2: Create/update the shipment
   515→      const body = {
   516→        date: form.date,
   517→        vehicleId: form.vehicleId,
   518→        description: form.description.trim() || undefined,
   519→        status: form.status,
   520→        orders: resolvedOrders,
   521→      };
   522→
   523→      if (editingShipment) {
   524→        const res = await fetch(`/api/shipments/${editingShipment.id}`, {
   525→          method: 'PUT',
   526→          headers: { 'Content-Type': 'application/json' },
   527→          body: JSON.stringify(body),
   528→        });
   529→        if (!res.ok) throw new Error('فشل في تحديث الشحنة');
   530→        toast.success('تم تحديث الشحنة بنجاح');
   531→      } else {
   532→        const res = await fetch('/api/shipments', {
   533→          method: 'POST',
   534→          headers: { 'Content-Type': 'application/json' },
   535→          body: JSON.stringify(body),
   536→        });
   537→        if (!res.ok) throw new Error('فشل في إضافة الشحنة');
   538→        toast.success('تم إضافة الشحنة بنجاح');
   539→      }
   540→
   541→      setDialogOpen(false);
   542→      fetchShipments();
   543→    } catch (err) {
   544→      toast.error(err instanceof Error ? err.message : 'حدث خطأ غير متوقع');
   545→    } finally {
   546→      setSubmitting(false);
   547→    }
   548→  }
