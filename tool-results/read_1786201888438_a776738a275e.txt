   101→function formatDate(dateStr: string): string {
   102→  const d = new Date(dateStr);
   103→  const y = d.getFullYear();
   104→  const m = String(d.getMonth() + 1).padStart(2, '0');
   105→  const day = String(d.getDate()).padStart(2, '0');
   106→  return `${y}/${m}/${day}`;
   107→}
   108→
   109→function formatDateInput(dateStr: string): string {
   110→  const d = new Date(dateStr);
   111→  const y = d.getFullYear();
   112→  const m = String(d.getMonth() + 1).padStart(2, '0');
   113→  const day = String(d.getDate()).padStart(2, '0');
   114→  return `${y}-${m}-${day}`;
   115→}

   241→  function removeOrder(index: number) {

   256→  async function handleSubmit(e: React.FormEvent) {
   257→    e.preventDefault();
   258→    if (!form.date) {
   259→      toast.error('التاريخ مطلوب');
   260→      return;
   261→    }
   262→    if (!form.vehicleId) {
   263→      toast.error('المركبة مطلوبة');
   264→      return;
   265→    }
   266→
   267→    // Validate orders
   268→    const validOrders = form.orders.filter((o) => o.clientId && o.packageCount > 0);
   269→    if (validOrders.length === 0) {
   270→      toast.error('يجب إضافة طلبية واحدة صالحة على الأقل');
   271→      return;
   272→    }
   273→
   274→    try {
   275→      setSubmitting(true);
   276→      const body = {
   277→        date: form.date,
   278→        vehicleId: form.vehicleId,
   279→        description: form.description.trim() || undefined,
   280→        status: form.status,
   281→        orders: validOrders.map((o) => ({
   282→          clientId: o.clientId,
   283→          packageCount: o.packageCount,
   284→          description: o.description?.trim() || undefined,
   285→        })),
   286→      };
   287→
   288→      if (editingShipment) {
   289→        const res = await fetch(`/api/shipments/${editingShipment.id}`, {
   290→          method: 'PUT',
   291→          headers: { 'Content-Type': 'application/json' },
   292→          body: JSON.stringify(body),
   293→        });
   294→        if (!res.ok) throw new Error('فشل في تحديث الشحنة');
   295→        toast.success('تم تحديث الشحنة بنجاح');
   296→      } else {
   297→        const res = await fetch('/api/shipments', {
   298→          method: 'POST',
   299→          headers: { 'Content-Type': 'application/json' },
   300→          body: JSON.stringify(body),
   301→        });
   302→        if (!res.ok) throw new Error('فشل في إضافة الشحنة');
   303→        toast.success('تم إضافة الشحنة بنجاح');
   304→      }
   305→
   306→      setDialogOpen(false);
   307→      fetchShipments();
   308→    } catch (err) {
   309→      toast.error(err instanceof Error ? err.message : 'حدث خطأ غير متوقع');
   310→    } finally {
   311→      setSubmitting(false);
   312→    }
   313→  }
