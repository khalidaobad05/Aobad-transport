   104→function formatDate(dateStr: string): string {
   105→  const d = new Date(dateStr);
   106→  const y = d.getFullYear();
   107→  const m = String(d.getMonth() + 1).padStart(2, '0');
   108→  const day = String(d.getDate()).padStart(2, '0');
   109→  return `${y}/${m}/${day}`;
   110→}

   145→// ========== Client Autocomplete Input ==========
   146→function ClientAutocomplete({
   147→  value,
   148→  onChange,
   149→  clients,
   150→}: {
   151→  value: string;
   152→  onChange: (clientId: string, clientName: string) => void;
   153→  clients: Client[];
   154→}) {
   155→  const [inputValue, setInputValue] = useState(value);
   156→  const [isOpen, setIsOpen] = useState(false);
   157→  const [highlightIndex, setHighlightIndex] = useState(-1);
   158→  const wrapperRef = useRef<HTMLDivElement>(null);
   159→  const inputRef = useRef<HTMLInputElement>(null);
   160→
   161→  useEffect(() => {
   162→    setInputValue(value);
   163→  }, [value]);
   164→
   165→  useEffect(() => {
   166→    function handleClickOutside(e: MouseEvent) {
   167→      if (wrapperRef.current && !wrapperRef.current.contains(e.target as Node)) {
   168→        setIsOpen(false);
   169→      }
   170→    }
   171→    document.addEventListener('mousedown', handleClickOutside);
   172→    return () => document.removeEventListener('mousedown', handleClickOutside);
   173→  }, []);
   174→
   175→  const filteredClients = inputValue.trim()
   176→    ? clients.filter((c) =>
   177→        c.name.toLowerCase().includes(inputValue.trim().toLowerCase())
   178→      )
   179→    : clients;
   180→
   181→  const shownClients = filteredClients.slice(0, 8);
   182→
   183→  const exactMatch = clients.find(
   184→    (c) => c.name.toLowerCase() === inputValue.trim().toLowerCase()
   185→  );
   186→
   187→  const isNewClient = inputValue.trim().length > 0 && !exactMatch;
   188→
   189→  function handleSelect(client: Client) {
   190→    setInputValue(client.name);
   191→    onChange(client.id, client.name);
   192→    setIsOpen(false);
   193→    setHighlightIndex(-1);
   194→  }
   195→
   196→  function handleInputChange(e: React.ChangeEvent<HTMLInputElement>) {
   197→    const val = e.target.value;
   198→    setInputValue(val);
   199→    onChange('', val);
   200→    setIsOpen(true);
   201→    setHighlightIndex(-1);
   202→  }
   203→
   204→  function handleAddNew() {
   205→    if (!inputValue.trim()) return;
   206→    onChange('', inputValue.trim());
   207→    setIsOpen(false);
   208→  }
   209→
   210→  function handleKeyDown(e: React.KeyboardEvent) {
   211→    const totalItems = shownClients.length + (isNewClient ? 1 : 0);
   212→
   213→    if (e.key === 'ArrowDown') {
   214→      e.preventDefault();
   215→      setHighlightIndex((prev) => (prev < totalItems - 1 ? prev + 1 : 0));
   216→    } else if (e.key === 'ArrowUp') {
   217→      e.preventDefault();
   218→      setHighlightIndex((prev) => (prev > 0 ? prev - 1 : totalItems - 1));
   219→    } else if (e.key === 'Enter') {
   220→      e.preventDefault();
   221→      if (highlightIndex >= 0 && highlightIndex < shownClients.length) {
   222→        handleSelect(shownClients[highlightIndex]);
   223→      } else if (highlightIndex === shownClients.length && isNewClient) {
   224→        handleAddNew();
   225→      }
   226→    } else if (e.key === 'Escape') {
   227→      setIsOpen(false);
   228→    } else if (e.key === 'Tab') {
   229→      if (shownClients.length === 1 && !isNewClient) {
   230→        e.preventDefault();
   231→        handleSelect(shownClients[0]);
   232→      } else if (exactMatch) {
   233→        e.preventDefault();
   234→        handleSelect(exactMatch);
   235→      }
   236→    }
   237→  }
   238→
   239→  return (
   240→    <div ref={wrapperRef} className="relative">
   241→      <div className="relative">
   242→        <Input
   243→          ref={inputRef}
   244→          type="text"
   245→          value={inputValue}
   246→          onChange={handleInputChange}
   247→          onFocus={() => setIsOpen(true)}
   248→          onKeyDown={handleKeyDown}
   249→          placeholder="اكتب اسم الزبون..."
   250→          className="w-full pr-8"
   251→          autoComplete="off"
   252→        />
   253→        {inputValue && (
   254→          <button
   255→            type="button"
   256→            onClick={() => {
   257→              setInputValue('');
   258→              onChange('', '');
   259→              inputRef.current?.focus();
   260→            }}
   261→            className="absolute left-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   262→          >
   263→            <X className="size-3.5" />
   264→          </button>
   265→        )}
   266→      </div>
   267→
   268→      {isOpen && (shownClients.length > 0 || isNewClient) && (
   269→        <div className="absolute z-50 top-full mt-1 w-full bg-white dark:bg-gray-900 border border-gray-200 dark:border-gray-700 rounded-lg shadow-lg max-h-48 overflow-y-auto">
   270→          {shownClients.map((client, idx) => (
   271→            <button
   272→              key={client.id}
   273→              type="button"
   274→              onClick={() => handleSelect(client)}
   275→              onMouseEnter={() => setHighlightIndex(idx)}
   276→              className={`
   277→                w-full text-right px-3 py-2 text-sm flex items-center gap-2 transition-colors
   278→                ${idx === highlightIndex
   279→                  ? 'bg-amber-50 dark:bg-amber-900/20 text-amber-900 dark:text-amber-100'
   280→                  : 'hover:bg-gray-50 dark:hover:bg-gray-800'
   281→                }
   282→              `}
   283→            >
   284→              <span className="truncate flex-1">{client.name}</span>
   285→              {exactMatch?.id === client.id && (
   286→                <Check className="size-3.5 text-emerald-500 shrink-0" />
   287→              )}
   288→            </button>
   289→          ))}
   290→          {isNewClient && (
   291→            <button
   292→              type="button"
   293→              onClick={handleAddNew}
   294→              onMouseEnter={() => setHighlightIndex(shownClients.length)}
   295→              className={`
   296→                w-full text-right px-3 py-2 text-sm flex items-center gap-2 border-t border-gray-100 dark:border-gray-800 transition-colors
   297→                ${highlightIndex === shownClients.length
   298→                  ? 'bg-emerald-50 dark:bg-emerald-900/20 text-emerald-700 dark:text-emerald-300'
   299→                  : 'hover:bg-gray-50 dark:hover:bg-gray-800 text-emerald-600 dark:text-emerald-400'
   300→                }
   301→              `}
   302→            >
   303→              <UserPlus className="size-3.5 shrink-0" />
   304→              <span>إضافة "{inputValue.trim()}" كزبون جديد</span>
   305→            </button>
   306→          )}
   307→        </div>
   308→      )}
   309→
   310→      {!exactMatch && inputValue.trim() && (
   311→        <p className="text-xs text-emerald-600 dark:text-emerald-400 mt-1 flex items-center gap-1">
   312→          <UserPlus className="size-3" />
   313→          سيتم إضافة "{inputValue.trim()}" إلى لائحة الزبائن
   314→        </p>
   315→      )}
   316→    </div>
   317→  );
   318→}

   621→  async function handleSubmit(e: React.FormEvent) {
   622→    e.preventDefault();
   623→    if (!form.date) {
   624→      toast.error('التاريخ مطلوب');
   625→      return;
   626→    }
   627→    if (!form.vehicleId) {
   628→      toast.error('المركبة مطلوبة');
   629→      return;
   630→    }
   631→
   632→    // Validate orders - each must have a client name and package count
   633→    const validOrders = form.orders.filter(
   634→      (o) => o.clientName.trim() && o.packageCount > 0
   635→    );
   636→    if (validOrders.length === 0) {
   637→      toast.error('يجب إضافة طلبية واحدة صالحة على الأقل (اسم الزبون وعدد الطرود)');
   638→      return;
   639→    }
   640→
   641→    try {
   642→      setSubmitting(true);
   643→
   644→      // Step 1: Ensure all clients exist
   645→      const resolvedOrders = await Promise.all(
   646→        validOrders.map(async (o) => {
   647→          let clientId = o.clientId;
   648→          if (!clientId) {
   649→            clientId = (await ensureClient(o.clientName.trim())) || '';
   650→          }
   651→          return {
   652→            clientId,
   653→            packageCount: o.packageCount,
   654→            price: o.price > 0 ? o.price : undefined,
   655→            description: o.description?.trim() || undefined,
   656→          };
   657→        })
   658→      );
   659→
   660→      const unresolved = resolvedOrders.find((o) => !o.clientId);
   661→      if (unresolved) {
   662→        throw new Error('فشل في تسجيل أحد الزبائن');
   663→      }
   664→
   665→      // Step 2: Create/update the shipment
   666→      const body = {
   667→        date: form.date,
   668→        vehicleId: form.vehicleId,
   669→        description: form.description.trim() || undefined,
   670→        status: form.status,
   671→        orders: resolvedOrders,
   672→      };
   673→
   674→      if (editingShipment) {
   675→        const res = await fetch(`/api/shipments/${editingShipment.id}`, {
   676→          method: 'PUT',
   677→          headers: { 'Content-Type': 'application/json' },
   678→          body: JSON.stringify(body),
   679→        });
   680→        if (!res.ok) throw new Error('فشل في تحديث الشحنة');
   681→        toast.success('تم تحديث الشحنة بنجاح');
   682→      } else {
   683→        const res = await fetch('/api/shipments', {
   684→          method: 'POST',
   685→          headers: { 'Content-Type': 'application/json' },
   686→          body: JSON.stringify(body),
   687→        });
   688→        if (!res.ok) throw new Error('فشل في إضافة الشحنة');
   689→        toast.success('تم إضافة الشحنة بنجاح');
   690→      }
   691→
   692→      setDialogOpen(false);
   693→      fetchShipments();
   694→    } catch (err) {
   695→      toast.error(err instanceof Error ? err.message : 'حدث خطأ غير متوقع');
   696→    } finally {
   697→      setSubmitting(false);
   698→    }
   699→  }
