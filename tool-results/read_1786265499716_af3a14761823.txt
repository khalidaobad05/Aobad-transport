   104→   104→function formatDate(dateStr: string): string {
   105→   105→  const d = new Date(dateStr);
   106→   106→  const y = d.getFullYear();
   107→   107→  const m = String(d.getMonth() + 1).padStart(2, '0');
   108→   108→  const day = String(d.getDate()).padStart(2, '0');
   109→   109→  return `${y}/${m}/${day}`;
   110→   110→}

   146→   146→function ClientAutocomplete({
   147→   147→  value,
   148→   148→  onChange,
   149→   149→  clients,
   150→   150→}: {
   151→   151→  value: string;
   152→   152→  onChange: (clientId: string, clientName: string) => void;
   153→   153→  clients: Client[];
   154→   154→}) {
   155→   155→  const [inputValue, setInputValue] = useState(value);
   156→   156→  const [isOpen, setIsOpen] = useState(false);
   157→   157→  const [highlightIndex, setHighlightIndex] = useState(-1);
   158→   158→  const wrapperRef = useRef<HTMLDivElement>(null);
   159→   159→  const inputRef = useRef<HTMLInputElement>(null);
   160→   160→
   161→   161→  useEffect(() => {
   162→   162→    setInputValue(value);
   163→   163→  }, [value]);
   164→   164→
   165→   165→  useEffect(() => {
   166→   166→    function handleClickOutside(e: MouseEvent) {
   167→   167→      if (wrapperRef.current && !wrapperRef.current.contains(e.target as Node)) {
   168→   168→        setIsOpen(false);
   169→   169→      }
   170→   170→    }
   171→   171→    document.addEventListener('mousedown', handleClickOutside);
   172→   172→    return () => document.removeEventListener('mousedown', handleClickOutside);
   173→   173→  }, []);
   174→   174→
   175→   175→  const filteredClients = inputValue.trim()
   176→   176→    ? clients.filter((c) =>
   177→   177→        c.name.toLowerCase().includes(inputValue.trim().toLowerCase())
   178→   178→      )
   179→   179→    : clients;
   180→   180→
   181→   181→  const shownClients = filteredClients.slice(0, 8);
   182→   182→
   183→   183→  const exactMatch = clients.find(
   184→   184→    (c) => c.name.toLowerCase() === inputValue.trim().toLowerCase()
   185→   185→  );
   186→   186→
   187→   187→  const isNewClient = inputValue.trim().length > 0 && !exactMatch;
   188→   188→
   189→   189→  function handleSelect(client: Client) {
   190→   190→    setInputValue(client.name);
   191→   191→    onChange(client.id, client.name);
   192→   192→    setIsOpen(false);
   193→   193→    setHighlightIndex(-1);
   194→   194→  }
   195→   195→
   196→   196→  function handleInputChange(e: React.ChangeEvent<HTMLInputElement>) {
   197→   197→    const val = e.target.value;
   198→   198→    setInputValue(val);
   199→   199→    onChange('', val);
   200→   200→    setIsOpen(true);
   201→   201→    setHighlightIndex(-1);
   202→   202→  }
   203→   203→
   204→   204→  function handleAddNew() {
   205→   205→    if (!inputValue.trim()) return;
   206→   206→    onChange('', inputValue.trim());
   207→   207→    setIsOpen(false);
   208→   208→  }
   209→   209→
   210→   210→  function handleKeyDown(e: React.KeyboardEvent) {
   211→   211→    const totalItems = shownClients.length + (isNewClient ? 1 : 0);
   212→   212→
   213→   213→    if (e.key === 'ArrowDown') {
   214→   214→      e.preventDefault();
   215→   215→      setHighlightIndex((prev) => (prev < totalItems - 1 ? prev + 1 : 0));
   216→   216→    } else if (e.key === 'ArrowUp') {
   217→   217→      e.preventDefault();
   218→   218→      setHighlightIndex((prev) => (prev > 0 ? prev - 1 : totalItems - 1));
   219→   219→    } else if (e.key === 'Enter') {
   220→   220→      e.preventDefault();
   221→   221→      if (highlightIndex >= 0 && highlightIndex < shownClients.length) {
   222→   222→        handleSelect(shownClients[highlightIndex]);
   223→   223→      } else if (highlightIndex === shownClients.length && isNewClient) {
   224→   224→        handleAddNew();
   225→   225→      }
   226→   226→    } else if (e.key === 'Escape') {
   227→   227→      setIsOpen(false);
   228→   228→    } else if (e.key === 'Tab') {
   229→   229→      if (shownClients.length === 1 && !isNewClient) {
   230→   230→        e.preventDefault();
   231→   231→        handleSelect(shownClients[0]);
   232→   232→      } else if (exactMatch) {
   233→   233→        e.preventDefault();
   234→   234→        handleSelect(exactMatch);
   235→   235→      }
   236→   236→    }
   237→   237→  }
   238→   238→
   239→   239→  return (
   240→   240→    <div ref={wrapperRef} className="relative">
   241→   241→      <div className="relative">
   242→   242→        <Input
   243→   243→          ref={inputRef}
   244→   244→          type="text"
   245→   245→          value={inputValue}
   246→   246→          onChange={handleInputChange}
   247→   247→          onFocus={() => setIsOpen(true)}
   248→   248→          onKeyDown={handleKeyDown}
   249→   249→          placeholder="اكتب اسم الزبون..."
   250→   250→          className="w-full pr-8"
   251→   251→          autoComplete="off"
   252→   252→        />
   253→   253→        {inputValue && (
   254→   254→          <button
   255→   255→            type="button"
   256→   256→            onClick={() => {
   257→   257→              setInputValue('');
   258→   258→              onChange('', '');
   259→   259→              inputRef.current?.focus();
   260→   260→            }}
   261→   261→            className="absolute left-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   262→   262→          >
   263→   263→            <X className="size-3.5" />
   264→   264→          </button>
   265→   265→        )}
   266→   266→      </div>
   267→   267→
   268→   268→      {isOpen && (shownClients.length > 0 || isNewClient) && (
   269→   269→        <div className="absolute z-50 top-full mt-1 w-full bg-white dark:bg-gray-900 border border-gray-200 dark:border-gray-700 rounded-lg shadow-lg max-h-48 overflow-y-auto">
   270→   270→          {shownClients.map((client, idx) => (
   271→   271→            <button
   272→   272→              key={client.id}
   273→   273→              type="button"
   274→   274→              onClick={() => handleSelect(client)}
   275→   275→              onMouseEnter={() => setHighlightIndex(idx)}
   276→   276→              className={`
   277→   277→                w-full text-right px-3 py-2 text-sm flex items-center gap-2 transition-colors
   278→   278→                ${idx === highlightIndex
   279→   279→                  ? 'bg-amber-50 dark:bg-amber-900/20 text-amber-900 dark:text-amber-100'
   280→   280→                  : 'hover:bg-gray-50 dark:hover:bg-gray-800'
   281→   281→                }
   282→   282→              `}
   283→   283→            >
   284→   284→              <span className="truncate flex-1">{client.name}</span>
   285→   285→              {exactMatch?.id === client.id && (
   286→   286→                <Check className="size-3.5 text-emerald-500 shrink-0" />
   287→   287→              )}
   288→   288→            </button>
   289→   289→          ))}
   290→   290→          {isNewClient && (
   291→   291→            <button
   292→   292→              type="button"
   293→   293→              onClick={handleAddNew}
   294→   294→              onMouseEnter={() => setHighlightIndex(shownClients.length)}
   295→   295→              className={`
   296→   296→                w-full text-right px-3 py-2 text-sm flex items-center gap-2 border-t border-gray-100 dark:border-gray-800 transition-colors
   297→   297→                ${highlightIndex === shownClients.length
   298→   298→                  ? 'bg-emerald-50 dark:bg-emerald-900/20 text-emerald-700 dark:text-emerald-300'
   299→   299→                  : 'hover:bg-gray-50 dark:hover:bg-gray-800 text-emerald-600 dark:text-emerald-400'
   300→   300→                }
   301→   301→              `}
   302→   302→            >
   303→   303→              <UserPlus className="size-3.5 shrink-0" />
   304→   304→              <span>إضافة "{inputValue.trim()}" كزبون جديد</span>
   305→   305→            </button>
   306→   306→          )}
   307→   307→        </div>
   308→   308→      )}
   309→   309→
   310→   310→      {!exactMatch && inputValue.trim() && (
   311→   311→        <p className="text-xs text-emerald-600 dark:text-emerald-400 mt-1 flex items-center gap-1">
   312→   312→          <UserPlus className="size-3" />
   313→   313→          سيتم إضافة "{inputValue.trim()}" إلى لائحة الزبائن
   314→   314→        </p>
   315→   315→      )}
   316→   316→    </div>
   317→   317→  );
   318→   318→}
