   621→   621→   621→  async function handleSubmit(e: React.FormEvent) {
   622→   622→   622→    e.preventDefault();
   623→   623→   623→    if (!form.date) {
   624→   624→   624→      toast.error('التاريخ مطلوب');

   626→   626→   626→    }
   627→   627→   627→    if (!form.vehicleId) {
   628→   628→   628→      toast.error('المركبة مطلوبة');
   629→   629→   629→      return;
   630→   630→   630→    }
   631→   631→   631→
   632→   632→   632→    // Validate orders - each must have a client name and package count
   633→   633→   633→    const validOrders = form.orders.filter(
   634→   634→   634→      (o) => o.clientName.trim() && o.packageCount > 0
   635→   635→   635→    );
   636→   636→   636→    if (validOrders.length === 0) {
   637→   637→   637→      toast.error('يجب إضافة طلبية واحدة صالحة على الأقل (اسم الزبون وعدد الطرود)');
   638→   638→   638→      return;
   639→   639→   639→    }
   640→   640→   640→
   641→   641→   641→    try {
   642→   642→   642→      setSubmitting(true);
   643→   643→   643→
   644→   644→   644→      // Step 1: Ensure all clients exist
   645→   645→   645→      const resolvedOrders = await Promise.all(
   646→   646→   646→        validOrders.map(async (o) => {
   647→   647→   647→          let clientId = o.clientId;
   648→   648→   648→          if (!clientId) {
   649→   649→   649→            clientId = (await ensureClient(o.clientName.trim())) || '';
   650→   650→   650→          }
   651→   651→   651→          return {
   652→   652→   652→            clientId,
   653→   653→   653→            packageCount: o.packageCount,
   654→   654→   654→            price: o.price > 0 ? o.price : undefined,
   655→   655→   655→            description: o.description?.trim() || undefined,
   656→   656→   656→          };
   657→   657→   657→        })
   658→   658→   658→      );
   659→   659→   659→
   660→   660→   660→      const unresolved = resolvedOrders.find((o) => !o.clientId);
   661→   661→   661→      if (unresolved) {
   662→   662→   662→        throw new Error('فشل في تسجيل أحد الزبائن');
   663→   663→   663→      }
   664→   664→   664→
   665→   665→   665→      // Step 2: Create/update the shipment
   666→   666→   666→      const body = {
   667→   667→   667→        date: form.date,
   668→   668→   668→        vehicleId: form.vehicleId,
   669→   669→   669→        description: form.description.trim() || undefined,
   670→   670→   670→        status: form.status,
   671→   671→   671→        orders: resolvedOrders,
   672→   672→   672→      };
   673→   673→   673→
   674→   674→   674→      if (editingShipment) {
   675→   675→   675→        const res = await fetch(`/api/shipments/${editingShipment.id}`, {
   676→   676→   676→          method: 'PUT',
   677→   677→   677→          headers: { 'Content-Type': 'application/json' },
   678→   678→   678→          body: JSON.stringify(body),
   679→   679→   679→        });
   680→   680→   680→        if (!res.ok) throw new Error('فشل في تحديث الشحنة');
   681→   681→   681→        toast.success('تم تحديث الشحنة بنجاح');
   682→   682→   682→      } else {
   683→   683→   683→        const res = await fetch('/api/shipments', {
   684→   684→   684→          method: 'POST',
   685→   685→   685→          headers: { 'Content-Type': 'application/json' },
   686→   686→   686→          body: JSON.stringify(body),
   687→   687→   687→        });
   688→   688→   688→        if (!res.ok) throw new Error('فشل في إضافة الشحنة');
   689→   689→   689→        toast.success('تم إضافة الشحنة بنجاح');
   690→   690→   690→      }
   691→   691→   691→
   692→   692→   692→      setDialogOpen(false);
   693→   693→   693→      fetchShipments();
   694→   694→   694→    } catch (err) {
   695→   695→   695→      toast.error(err instanceof Error ? err.message : 'حدث خطأ غير متوقع');
   696→   696→   696→    } finally {
   697→   697→   697→      setSubmitting(false);
   698→   698→   698→    }
   699→   699→   699→  }
